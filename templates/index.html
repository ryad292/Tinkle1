/* ========================================================= TINKLE ALWAYS-ON VOICE / WAKE WORD ========================================================= */ const socket = io(); let recognition = null; let isRecognizing = false; let shouldRecognize = true; let isSpeaking = false; /* ========================================================= VOICE ========================================================= */ let selectedVoice = null; function loadTinkleVoice() { const voices = speechSynthesis.getVoices(); if (!voices.length) return; const preferred = [ "Microsoft David", "Microsoft Mark", "Google UK English Male", "Google US English", "Alex", "Daniel", "Arthur", "Thomas", "Fred" ]; selectedVoice = voices.find(v => preferred.some(name => v.name.toLowerCase().includes( name.toLowerCase() ) ) ); if (!selectedVoice) { selectedVoice = voices.find(v => /male|man|david|mark|alex|daniel|arthur|thomas/i .test(v.name) ); } if (!selectedVoice) { selectedVoice = voices.find(v => v.lang && v.lang.toLowerCase().startsWith("en") ); } } speechSynthesis.onvoiceschanged = loadTinkleVoice; loadTinkleVoice(); /* ========================================================= TINKLE SPEAK ========================================================= */ function tinkleSpeak(text) { if (!text) return; isSpeaking = true; speechSynthesis.cancel(); loadTinkleVoice(); const message = new SpeechSynthesisUtterance(text); if (selectedVoice) message.voice = selectedVoice; message.rate = 0.90; message.pitch = 0.75; message.volume = 1.0; message.onstart = () => { window.tinkleStartSpeaking(); }; message.onend = () => { isSpeaking = false; window.tinkleStopSpeaking(); restartTinkleListening(); }; message.onerror = () => { isSpeaking = false; window.tinkleStopSpeaking(); restartTinkleListening(); }; speechSynthesis.speak(message); } /* ========================================================= SPEECH RECOGNITION ========================================================= */ const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { console.error( "Speech Recognition is not supported." ); window.tinkleError( "VOICE RECOGNITION NOT SUPPORTED" ); } else { recognition = new SpeechRecognition(); /* * Continuous listening */ recognition.continuous = true; recognition.interimResults = false; recognition.maxAlternatives = 1; /* * English wake word */ recognition.lang = "en-US"; /* ============================================= START ============================================= */ recognition.onstart = () => { isRecognizing = true; console.log( "TINKLE LISTENING..." ); }; /* ============================================= AUTOMATIC RESTART ============================================= */ recognition.onend = () => { isRecognizing = false; if ( shouldRecognize && !isSpeaking ) { setTimeout( restartTinkleListening, 300 ); } }; /* ============================================= ERROR ============================================= */ recognition.onerror = (event) => { console.log( "Tinkle voice error:", event.error ); isRecognizing = false; if ( event.error === "not-allowed" ) { window.tinkleError( "MICROPHONE PERMISSION REQUIRED" ); return; } if ( event.error === "service-not-allowed" ) { window.tinkleError( "VOICE SERVICE NOT AVAILABLE" ); return; } if (shouldRecognize) { setTimeout( restartTinkleListening, 1000 ); } }; /* ============================================= HEARD ============================================= */ recognition.onresult = function(event) { const result = event.results[ event.results.length - 1 ]; if (!result) return; const transcript = result[0] .transcript .trim(); if (!transcript) return; console.log( "TINKLE HEARD:", transcript ); const text = transcript.toLowerCase(); /* ========================================= WAKE WORD ========================================= */ const wakeWord = text.includes("tinkle") || text.includes("تينكل") || text.includes("تنكل"); if (!wakeWord) return; console.log( "TINKLE WAKE WORD DETECTED" ); /* * Update interface */ window.tinkleStartListening(); /* * Stop recognition while Tinkle speaks. * This prevents Tinkle from hearing itself. */ try { recognition.stop(); } catch (e) {} /* * Wake response */ setTimeout(() => { tinkleSpeak( "Yes, Doctor. I'm listening." ); socket.emit( "wake_word_detected", { status: "active", transcript: transcript } ); }, 150); }; /* ============================================= AUTOMATIC LISTENING ============================================= */ function restartTinkleListening() { if (!recognition) return; if (!shouldRecognize) return; if (isRecognizing) return; if (isSpeaking) return; try { recognition.start(); console.log( "TINKLE AUTO LISTENING STARTED" ); } catch (error) { setTimeout( restartTinkleListening, 1000 ); } } /* * Start automatically. */ setTimeout( restartTinkleListening, 1000 ); } /* ========================================================= MANUAL VOICE CONTROL ========================================================= */ window.tinkleStartVoice = function() { shouldRecognize = true; restartTinkleListening(); }; window.tinkleStopVoice = function() { shouldRecognize = false; if (recognition) { try { recognition.stop(); } catch (e) {} } speechSynthesis.cancel(); }; /* ========================================================= TEST ========================================================= */ window.tinkleTestVoice = function() { tinkleSpeak( "Yes, Doctor. Tinkle is online." ); };
